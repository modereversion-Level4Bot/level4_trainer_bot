 Сделать чистый ZIP проекта
$src = (Get-Location).Path
$dst = Join-Path $env:TEMP "level4_trainer_bot_current_clean"
$out = "C:\Users\Константин Хотеев\Desktop\MVP v1.0\level4_trainer_bot_CURRENT_CLEAN.zip"

Remove-Item $dst -Recurse -Force -ErrorAction SilentlyContinue
Remove-Item $out -Force -ErrorAction SilentlyContinue

New-Item -ItemType Directory -Path $dst | Out-Null

robocopy $src $dst /E `
  /XD ".venv" "venv" "__pycache__" ".git" ".mypy_cache" ".pytest_cache" "secrets" "backups" `
  /XF ".env" "*.db" "*.sqlite" "*.sqlite3" "*.pyc" "*.pyo" "*.log" "*.zip" "google_service_account.json"

Compress-Archive -Path "$dst\*" -DestinationPath $out -Force

Remove-Item $dst -Recurse -Force

Write-Host "Created: $out"
